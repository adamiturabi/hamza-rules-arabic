\section{\texorpdfstring{\foreignlanguage{arabic}{\altfamily المقدمة}}{المقدمة}}\label{ux627ux644ux645ux642ux62fux645ux629}

\begin{otherlanguage}{arabic}

\altfamily 

بسم الله الرحمن الرحيم

الحمد الله والصلاة والسلام على نبينا محمد. أما بعد:

اعلم رحمك الله أن الألف تكون أحيانًا
\href{https://ar.wikipedia.org/wiki/\%D8\%B5\%D8\%A7\%D8\%A6\%D8\%AA}{صائتًا}
وأحيانًا
\href{https://ar.wikipedia.org/wiki/\%D8\%B5\%D8\%A7\%D9\%85\%D8\%AA}{صامتًا}.
وحين تكون صامتا تلفظ بصوت الهمزة. والهمزة تحرك ويسكن عليها كما يُفعل
بسائر الصوامت. وكان حقها أن تُكتب بالألف فذهبنا نكتب:

\begin{itemize}
\tightlist
\item
  هـٰأُلاإِ (هؤلاءِ)
\item
  تَوْأَم (تَوْءَم)
\item
  خَطِيأَة (خَطِيءَة)
\item
  شَيْأَا (شَيْءَا)
\item
  سُأَال (سؤال)
\item
  رأس
\item
  بِأْسَ (بئس)
\item
  سُأْل (سُؤْل)
\end{itemize}

ولكن بعض القبائل أشهرها قريش كانوا لا يلفظون الهمزة إلا قليلًا. فيبدلونها
مدًّا أو حركة. فنظنهم يقولونَ ويكتبون:

\begin{itemize}
\tightlist
\item
  هـٰوُلا (هؤلاء)
\item
  تَوَّم (تَوْءَم)
\item
  خَطِيَّة (خَطِيءَة)
\item
  شَيَّا (شَيْءَا)
\item
  سُوَال (سؤال)
\item
  رَاس (رأس)
\item
  بِيسَ (بئس)
\item
  سُول (سُؤْل)
\end{itemize}

واتفق أن تَمَازُجًا بين كتابة لغة القريش غير المهموزة وتلفُّظ لغات العرب
المهموزة صار المعيار للعربية التي كانت يكتب بها في الكتب وينطق بها في
الخُطَب. والمعيار لا حرج في اقتدائه لا سيما لصلتنا بتراثنا القيم.

فشكّل الكتبة القدماء للهمزة تشكيلًا «ء» يشبه العين. وكتبوها على كراسي أو
دون كرسي على حسب قواعد يعلمونها.

ولكن يشكل علينا اليوم كتابة الهمزة. متى نجعل لها كرسيًّا فنكتبها مدعومة
على الألف أو الياء أو الواو؟ ومتى نتركها مستغنية فنكتبها دون كرسي؟

فها أنا ذا أقدم لكم قواعد تبسط الإشكال إن شاء الله:

\end{otherlanguage}

\section{\texorpdfstring{\foreignlanguage{arabic}{\altfamily الهمزة
المبتدأة}}{الهمزة المبتدأة}}\label{ux627ux644ux647ux645ux632ux629-ux627ux644ux645ux628ux62aux62fux623ux629}

\begin{otherlanguage}{arabic}

\altfamily 

أما إن كانت حركة الهمزة المبتدأة ممدودة بالألف فتكتب الهمزة مستغنية دون
كرسي. فنكتب «ءَاخِرة». وكما قلنا آنفا جمع بين الهمزة والألف في غير المصحف
وكتب «آخرة».

وأما إن كانت حركة الهمزة غير ممدودة بألف فتكتب الهمزة المبتدأة على
الألف. فنكتب «أَمر» و«إسلام» و«أُرِيدُ» و«أُوخِذَ» و«إِيمان».

وإن سبق الهمزةَ حرفٌ أو ما يزيد منه فيتصل بالحرف والهمزة، بقيت الهمزة على
الألف. فنكتب «بِأَمْر» و«الإسلام» و«فأُريدُ» و«الءاخرة» («الآخرة»).

ويستثنى من هذه القاعدة بضع لفظات تعامل كأن همزتهن متوسطة مثل «لِئَلَّا»
و«لَئِنْ».

\end{otherlanguage}

\section{\texorpdfstring{\foreignlanguage{arabic}{\altfamily الهمزة
المتوسطة}}{الهمزة المتوسطة}}\label{ux627ux644ux647ux645ux632ux629-ux627ux644ux645ux62aux648ux633ux637ux629}

\begin{otherlanguage}{arabic}

\altfamily 

وهي أهم وأشد إشكالا.

احفظ هذه الأصول:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\item
  الكسر أقوى من الضم والضم أقوى من الفتح. والسكون هو الأضعف.
\item
  الحركة الممدودة أقوى من نفس الحركة غير الممدودة.
\item
  المدود «ــِي وــُو وــَا» وما يشبها «ــَوْ وــَيْ» والسكون يفضل لهن كتابة
  الهمزة مستغنية دون كرسي. وأما الحركات غير الممدودة فيفضل للكسر كتابة
  الهمزة مدعومة على كرسيها الياء، وللضم على الواو، وللفتح على الألف.
\end{enumerate}

فمن هذه الأصول نضع جدولًا للصوائت أول صفوفه لأقواها وآخر صفوفه لأضعفها:

{\def\LTcaptype{none} % do not increment counter
\begin{longtable}[]{@{}rr@{}}
\toprule\noalign{}
الصائت & الكتابة \\
\midrule\noalign{}
\endhead
\bottomrule\noalign{}
\endlastfoot
المد بالياء وشبهه «ــِي/ــَيْ» & ء \\
الكسر غير الممدود «ــِـ» & ئ \\
المد بالواو وشبهه «ــُو/ــَوْ» & ء \\
الضم غير الممدود «ــُـ» & ؤ \\
المد بالألف «ــَا» & ء \\
الفتح غير الممدود «ــَـ» & أ \\
السكون «ــْـ» & ء \\
\end{longtable}
}

وإما تستعلمن كتابة الهمزة في لفظة فاجعل الصائت الذي للهمزة والصائت الذي
\textbf{قبل} الهمزة في الجدول الموضوع آنفا. فانظر أيهما أقوى. واكتب
الهمزة كتابة الصائت الأقوى.

واعلم أيضا:

\begin{itemize}
\tightlist
\item
  إن كانت الهمزة تكتب مستغنية بدون كرسي وهي بين حرفين متواصلين (يعني لم
  يكن الحرف الأول ألفًا أو دالا أو راء أو واوا، مهملة كانت أم معجمة) فلا
  يقطع تواصلهما. بل تكتب الهمزة بين الحرفين الأول والآخر، فوق الخط الذي
  يتصل به بين الحرفين هكذا «ـءـ».
\item
  أن الهمزة المفتوحة المستغنية بعدها ألف (أعني «ءَا» أو «ـءَا») تكتب «آ»
  في غير المصحف إن كانت الألف أصلية.
\end{itemize}

لنمارس ونر كتابة الهمزة في اللفظات:

مثالنا الأول «خَطِيءَة». نقايِس بين الصائت الذي على الهمزة وهو الفتح والصائت
قبلها وهو المد بالياء. ننظر في الجدول أن المد بالياء أقوى وكتابتها
المفضلة همزة مستغنية دون كرسي. ونرى أن الياء قبل الهمزة يتصل بينها وبين
التاء المربوطة بعدها. فلا نقطعن وصْلهما. فنكتب الهمزة دون كرسي فوق الخط
الذي يتصل به بين الحرفين هكذا «خَطِيءَة».

قس على ذلك واكتب «تَوْءَم» و«مَوْئِل» و«مَسْءُول» و«نَشْأَة» و«المَرْأَة» و«أَفْئِدَة»
و«سُؤَال» و«سَائِل» و«سَأَلَ» و«سُئِلَ» و«يَئِسَ» و«مُطْمَئِنّ» و«سَيِّئَة» و«رَأْس» و«بِئْسَ»
و«سُؤْل» و«رُءُوس» و«شُءُون» و«شَنَآن» و«ظَمْآن» و«أُولَـٰئِكَ» و«سَءَّال» و«لَءَّال»
و«اسْتِيءَاس» و«رَأَى»

وانظر في «سَءَّال» و«لَءَّال» لم نجمع بين الهمزة المستغنية والألف فلم نكتب
«آ». وهذا لوقوع الشَّد الذي يمنع ذلك.

وفي «لَءَّال» بائع اللؤلؤ، لم نأذن للهمزة أن تقطع وصل اللام والألف الخاص
«لا». فلم نكتب «لَـءَّـال» وهو خطأ.

و«اسْتِيءَاس» لم نكتبها «اسْتِيآس» لقبح شكلها هكذا وهذا رأي منا يؤخذ أو يترك.

وأما «رَأَى» فصائت الهمزة هو الفتح الممدود ولكن بغير الألف بل بألف على
صورة ياء. فصائته الذي نجعل في الجدول هو الفتح غير الممدود الذي يساوي
الصائت قبل الهمزة فتكتب الهمزة على الألف.

ولفظة «مائة» شاذة فلك فيها خيار. إما أن تكتبها كما نقلت وإما أن تكتبها
«مئة» على القياس. ولا نرى بأسًا في أي منهما.

\end{otherlanguage}

\section{\texorpdfstring{\foreignlanguage{arabic}{\altfamily الهمزة
المتطرفة}}{الهمزة المتطرفة}}\label{ux627ux644ux647ux645ux632ux629-ux627ux644ux645ux62aux637ux631ux641ux629}

\begin{otherlanguage}{arabic}

\altfamily 

قواعد الهمزة المتطرفة مثل قواعد الهمزة المتوسطة إلا لا ينظر إلى حركة
الهمزة. وإنما ينظر إلى الصائت الذي يقع قبل الهمزة. يوضع الصائت في الجدول
وتكتب الهمزة كما يشير إليه الجدول. إليك بأمثلة:

«يَقْرَأُ» و«بَطُؤَ» و«متكِئ» و«شَيْء» و«ضَوْء» و«سُوء» و«سَيِّئ» و«مبتدَأ» و«دعاء»
و«جِيء» و«عِبْء»

ويستثنى من القياس لفظة «تَبَوُّء» فتكتب الهمزة مستغنية لا على واو ثانية.

وإذا كتبت الهمزة المتطرفة على الألف والهمزة مكسورة فلك فيها خيار. إما أن
تكتب الهمزة تحت الألف وإما أن تكتبها فوقها. مثل «من المَبْدَإِ» أو «من
المبدأ». ونفضل أن نكتبها فوق الألف إلا إذا شكّل بالكسرة تحت الألف.

\end{otherlanguage}

\section{\texorpdfstring{\foreignlanguage{arabic}{\altfamily الهمزة
المتطرفة الملحقة
بمزيد}}{الهمزة المتطرفة الملحقة بمزيد}}\label{ux627ux644ux647ux645ux632ux629-ux627ux644ux645ux62aux637ux631ux641ux629-ux627ux644ux645ux644ux62dux642ux629-ux628ux645ux632ux64aux62f}

\begin{otherlanguage}{arabic}

\altfamily 

قد يكون همزة متطرفة أصلًا ويلحق بمزيد مثل «مَجِيءُهُ» و«خاطِءين».

الأصل في هذه الحالات أن تحسب الهمزة متوسطة وتتبع قواعد الهمزة المتوسطة.
فتكتب:

«بَرِيءُونَ» و«بَرِيءَانِ» و«بَرِيءِينَ» و«بَرِيءَيْنِ» و«سِيءَتْ» و«شَيْءُهُ» و«شَيْءَهُ» و«شَيْءِهِ»
و«شَيْءَانِ» و«شَيْءَيْنِ» و«مَجِيءُهُ» و«مَجِيءَهُ» و«مَجِيءِهِ» و«سُوئِهِ» و«ضَوْئِهِ» و«يَسُوءُونَ»
و«سُوءُهُ» و«سُوءَهُ» و«سُوءَانِ» و«ضَوْءَهُ» و«ضَوْءَانِ» و«مُرْجَءَات» و«خَاطِءِين» و«خَاسِءِين»
و«مُتَّكِءِينَ» و«مُتَّكِئُونَ» و«يُبَرِّئُونَ» و«يُبَرَّءُونَ» و«ٱقْرَءُوا» و«سَيِّئَات» و«فِئَات»
و«فِئُون» و«مِئَات» و«مِئُون» و«يُطْفِئُوا» و«مُسْتَهْزِئُون» و«مُنْشِئُون» و«يُضَاهِئُونَ»
و«مُرْجِئُون» و«قِرَاءَات»

انظر كيف لا يجتمع «ءَا» فيكون «آ». هذا لأن الألف ليست أصلية.

يستثنى من الأصل الهمزة التي قبلها ساكن فتكتب مستغنية ولو ويلحق بمزيد.
مثل:

«جُزْءَانِ» و«عِبْءَانِ» و«عِبْءَيْنِ» و«بُطْءَهُ» و«بُطْءُهُ» و«بُطْءِهِ»

\end{otherlanguage}

\section{\texorpdfstring{\foreignlanguage{arabic}{\altfamily الهمزة
المنونة}}{الهمزة المنونة}}\label{ux627ux644ux647ux645ux632ux629-ux627ux644ux645ux646ux648ux646ux629}

\begin{otherlanguage}{arabic}

\altfamily 

الهمزة المنونة متطرفة لا إشكال فيها إلا التي تنون تنوين الفتح فتكتب
بعدها ألف (إلا إذا كان قبلها المد بالألف). فإذا كتبت بعد الهمزة المنونة
ألف حسبت همزة متوسطة وبعدها المد بالألف. كأننا نقف عليها. ويوضع في
الجدول الصائت الذي قبل الهمزة وصائت الهمزة، وهو المد بالألف فينظر أيهما
أقوى. كما فعلنا من قبل.

فلنأخذ «مبتدَأ» مثالا. لا مد بالألف قبل الهمزة فإذا نونّا الهمزة ألحقناها
بألف. فالصائت الذي قبل الهمزة هي حركة الدال وهو الفتح. وصائت الهمزة هو
المد بالألف. المد بالألف أقوى فتكتب الهمزة مستغنية وبعدها ألف هكذا
«مبتدَءًا».

كذلك «شَيْءًا» و«مَلْجَءًا» و«جُزْءًا» و«سَيِّئًا»

\end{otherlanguage}

\section{\texorpdfstring{\foreignlanguage{arabic}{\altfamily اختلافات
قديمة
وحديثة}}{اختلافات قديمة وحديثة}}\label{ux627ux62eux62aux644ux627ux641ux627ux62a-ux642ux62fux64aux645ux629-ux648ux62dux62fux64aux62bux629}

\begin{otherlanguage}{arabic}

\altfamily 

القواعد التي وضعناها هي محاولة إلى كتابة قريش. ولنا أن نقس بها فنكتب أي
لفظة شئنا.

ولكن إذا رأينا في المصحف الذي كتب بالرسم العثماني رأينا بعض لفظات كتبت
على خلاف القياس الذي وضعناه.

فنرى فيه «يُضَاهِءُونَ» و«مُتَّكِءُونَ» و«مُتَّكِءِينَ» و«سَيِّءَات» و«يُطْفِءُوا» و«مُرْجِءُونَ»
و«مُسْتَهْزِءُونَ»

وفي كل منهن كتبت الهمزة مستغنية دون كرسي. فإذا قسنا قياسنا كتبنا الهمزات
فيهن على الياء مثل «يُضَاهِئُونَ». أفأخطأنا؟ وما الصواب؟

لا نرى بأسًا أولًّا في اتباع قياسنا. ولكن من أراد أن يلتزم بكتابة قريش
التزامًا شديدا فليعدل القواعد هكذا:

إذا كانت الهمزة متطرفة أصلا تلحق بمد مزيد، فتكتب الهمزة مستغنية.

ولكن لا تفعل هذا إن كانت اللفظة منقوصة، مثل «مِئُونَ» و«فِئُون».

وتعارض آخر: نرى في المصحف «أَفْءِدَة» ولو قسنا قياسنا كتبناها «أَفْئِدَة». ولكن
«نَشْأَة» هكذا كتبت لا «نَشْءَة». فلا نرى كما قلنا بأسًا في قياسنا ولو اختلف
أحيانًا بكتابة قريش.

ودون ذلك، اختلف قياسنا والقواعد الحديثة التي وضعتها المجامع وغيرها
تيسيرًا. فبعض هذه الاختلافات لا نرى بأسًا فيها (ولو لم تر احتياجا لها) مثل
«مسؤول» دون «مسءول» و«رَئيس» دون «رءيس». وسبب إجازتنا إياها هو إن بدلنا
الهمزة بكرسيها صلحت اللفظة أو كادت تصلح تلفُّظَ اللغة غير المهموزة. مثلا:
«مَسْوُول» و«رَيِيس». فمن أراد أن يكتب واوين وياءين في مثل هذه اللفظات
فليفعل.

ولكن بعض القواعد الحديثة نعارضها أشد معارضة. أقبحها تحويل الهمزة
المستغنية بين حرفين متواصلين إلى همزة على كرسي يائي. مثل «مَسْئُول» دون
«مَسْءُول» و«شُئُون» دون «شُءُون» و«سَئَّال» دون «سَءَّال». لا داعي لمثل هذه
التيسيرات. فلو بدلنا الهمزة في هذه التيسيرات ياءً لظهر التفاهت. انظر إلى
فساد الأمر في «مَسْيُول» و«شُيُون» و«سيَّال».

وما زاد الأمر سوءًا أن معظم الخطوط الحاسوبية لا يجيز كتابة همزة مستغنية
بين حرفين متواصلين إلا أن تقطع الهمزة تواصلهما.

قلما صلح خط حاسوبي في هذا الأمر. أحسنها الخط
\href{https://github.com/aliftype/amiri}{«الأميري»} الذي كتبت به هذه
المقالة.

وخط آخر أحسن من الأخريات خط
\href{https://github.com/nuqayah/kitab-font}{«الكتاب»} ولكن تضطر فيها
إلى أن تكتب الهمزة فوق «تطويل».

ونسأل الله أن يوفقنا لإصلاح الأمر.
